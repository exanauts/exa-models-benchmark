% preview.tex — renders every generated artifact THE WAY THE PAPER RENDERS IT:
% same class, same geometry, and the same float wrapping each table gets in
% sections/. Inputting a bare tabular shows you the tabular, not the table --
% hardware overflowed here only because the paper sets \tabcolsep=4pt around it
% and this file did not, and sgm_summary looked wrong because the paper wraps it
% in \resizebox{\textwidth}.
%
% Build:  make review        (regenerates build/_artifacts.tex, then compiles)
% Output: data/preview.pdf   — not deployed; `make deploy` is the separate step.
%
% The artifact list is globbed into build/_artifacts.tex, so a renamed or new
% artifact appears without editing this file.
\documentclass[11pt]{article}

% The paper's own preamble: class options, geometry, fonts, table and pgfplots
% setup. Inputting it rather than restating it is the point -- a restated
% preamble drifts, and then the preview stops predicting the paper.
\input{../../preamble.tex}

% The generated wrappers include their panels from \figdir; point it at
% build/ so the preview renders exactly the file the paper inputs.
\newcommand{\figdir}{build/figures}

\setlength{\parindent}{0pt}

% TOC entries must not be typeset in math mode. \detokenize alone is not enough:
% the entry is written to the .toc file and re-expanded on the next run, so the
% underscores came back as subscripts (gpu_s ummary_c ops). \protect keeps the
% pieces intact through that round trip.
% #1 = file stem (used to build the path), #2 = display name with underscores
% already escaped by the generator. Letting TeX sanitise #1 for the .toc failed:
% \detokenize is written to the file and re-expanded, and the underscores came
% back as math subscripts (gpu_s ummary_c ops).
% Numbered (not starred) so every artifact carries a subsection number and the
% TOC entry is generated by \subsection itself rather than hand-added.
\newcommand{\artifacthead}[2]{\subsection{\texttt{#2}}}

% Each table gets the wrapping the paper actually gives it, so width and
% crowding here mean what they will mean in the paper:
%   hardware      sections/numerics.tex — \setlength{\tabcolsep}{4pt}
%   sgm_summary   sections/numerics.tex — \resizebox{\textwidth}{!}
%   gpu_summary_* results/tables/gpu_summary.tex — plain, inside a table float
%   results_*     sections/appendix.tex — longtable, already its own float
% Each table gets the wrapping the paper actually gives it. Taken from the
% paper, not invented:
%   hardware      sections/numerics.tex  -> table float, \tabcolsep 4pt
%   sgm_summary   sections/numerics.tex  -> table float, \resizebox{\textwidth}
%   gpu_summary_* results/tables/gpu_summary.tex -> table float + caption
%   results_*     sections/appendix.tex  -> BARE \input, no float
%
% The last one matters most and is what was wrong: results_* are longtable, and
% a longtable inside a table float cannot break across pages, so the "continued
% on next page" machinery the paper relies on never appears. The paper inputs
% them bare; so does this now.
\newcommand{\artifacttable}[2]{%
  \artifacthead{#1}{#2}%
  \ifnum\pdfstrcmp{#1}{hardware}=0
    \begin{table}[H]\centering{\setlength{\tabcolsep}{4pt}\input{build/tables/#1.tex}}\end{table}%
  \else\ifnum\pdfstrcmp{#1}{sgm_summary}=0
    \begin{table}[H]\centering\resizebox{\textwidth}{!}{\input{build/tables/#1.tex}}\end{table}%
  \else\ifnum\pdfstrcmp{#1}{gpu_summary_lv}=0
    \begin{table}[H]\centering\caption{GPU speedup over single-threaded CPU: Luk\v{s}an--Vl\v{c}ek (SGM, $\sigma = 10^{-5}$\,s). \textsuperscript{*}Float32.}\input{build/tables/#1.tex}\end{table}%
  \else\ifnum\pdfstrcmp{#1}{gpu_summary_cops}=0
    \begin{table}[H]\centering\caption{GPU speedup over single-threaded CPU: COPS (SGM, $\sigma = 10^{-5}$\,s). \textsuperscript{*}Float32.}\input{build/tables/#1.tex}\end{table}%
  \else\ifnum\pdfstrcmp{#1}{gpu_summary_opf_polar}=0
    \begin{table}[H]\centering\caption{GPU speedup over single-threaded CPU: PGLIB-OPF (polar) (SGM, $\sigma = 10^{-5}$\,s). \textsuperscript{*}Float32.}\input{build/tables/#1.tex}\end{table}%
  \else\ifnum\pdfstrcmp{#1}{gpu_summary_opf_rect}=0
    \begin{table}[H]\centering\caption{GPU speedup over single-threaded CPU: PGLIB-OPF (rectangular) (SGM, $\sigma = 10^{-5}$\,s). \textsuperscript{*}Float32.}\input{build/tables/#1.tex}\end{table}%
  \else
    % results_* and anything else longtable-based: bare, exactly as the paper does
    \input{build/tables/#1.tex}%
  \fi\fi\fi\fi\fi\fi
  \clearpage}
